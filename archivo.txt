public class EjemploJava {
    public static void main(String[] args) {
        String nombreJava = "Carlos";
        String apellidoJava = "Perez";
        int edadJava = 20;
        boolean activoJava = true;
        String programaJava = "ADSO";
        double[] notasJava = {4.2, 3.9, 4.7, 5.0};
        System.out.println(nombreJava);
        System.out.println(apellidoJava);
        System.out.println(edadJava);
        System.out.println(activoJava);
        System.out.println(programaJava);
        double sumaJava = 0;
        for (double nota : notasJava) {
            sumaJava += nota;
            System.out.println(nota);
        }
        double promedioJava = sumaJava / notasJava.length;
        System.out.println(promedioJava);
        if (promedioJava >= 3) {
            System.out.println("Aprobado");
        } else {
            System.out.println("Reprobado");
        }
        for (int i = 0; i < notasJava.length; i++) {
            System.out.println(i + " " + notasJava[i]);
        }
        String saludo = saludarJava(nombreJava);
        System.out.println(saludo);
        int[] cuadradosJava = new int[5];
        for (int numero = 1; numero <= 5; numero++) {
            cuadradosJava[numero - 1] = numero * numero;
        }
        for (int valor : cuadradosJava) {
            System.out.println(valor);
        }
        System.out.println("Fin del bloque Java");
    }
    public static String saludarJava(String nombre) {
        return "Hola " + nombre;
    }
}

fn saludar_rust(nombre: &str) -> String {
    format!("Hola {}", nombre)
}
fn main() {
    let nombre_rust = "Sofia";
    let apellido_rust = "Lopez";
    let edad_rust = 22;
    let activo_rust = true;
    let programa_rust = "ADSO";
    let notas_rust = vec![4.1, 3.7, 4.9, 5.0];
    println!("{}", nombre_rust);
    println!("{}", apellido_rust);
    println!("{}", edad_rust);
    println!("{}", activo_rust);
    println!("{}", programa_rust);
    println!("{:?}", notas_rust);
    let mut suma_rust = 0.0;
    for nota in &notas_rust {
        suma_rust += nota;
        println!("{}", nota);
    }
    let promedio_rust = suma_rust / notas_rust.len() as f64;
    println!("{}", promedio_rust);
    if promedio_rust >= 3.0 {
        println!("Aprobado");
    } else {
        println!("Reprobado");
    }
    for (indice, nota) in notas_rust.iter().enumerate() {
        println!("{} {}", indice, nota);
    }
    let saludo_rust = saludar_rust(nombre_rust);
    println!("{}", saludo_rust);
    let mut cuadrados_rust = Vec::new();
    for numero in 1..=5 {
        cuadrados_rust.push(numero * numero);
    }
    println!("{:?}", cuadrados_rust);
    println!("Privet mir");
    println!("Moskva");
    println!("Cirilico");
    println!("Fin del bloque Rust");
}

#include <iostream>
#include <vector>
using namespace std;
int main() {
    string nombreCpp = "Mateo";
    int edadCpp = 19;
    bool activoCpp = true;
    vector<double> notasCpp = {4.0, 3.6, 4.8, 5.0};
    cout << nombreCpp << endl;
    cout << edadCpp << endl;
    cout << activoCpp << endl;
    double sumaCpp = 0;
    for (double nota : notasCpp) {
        sumaCpp += nota;
        cout << nota << endl;
    }
    double promedioCpp = sumaCpp / notasCpp.size();
    cout << promedioCpp << endl;
    if (promedioCpp >= 3) {
        cout << "Aprobado" << endl;
    } else {
        cout << "Reprobado" << endl;
    }
    for (int i = 0; i < notasCpp.size(); i++) {
        cout << i << " " << notasCpp[i] << endl;
    }
    cout << "Fin del bloque C++" << endl;
    return 0;
}

package main
import "fmt"
func main() {
    nombreGo := "Camila"
    edadGo := 23
    activoGo := true
    notasGo := []float64{4.2, 3.8, 4.9, 5.0}
    fmt.Println(nombreGo)
    fmt.Println(edadGo)
    fmt.Println(activoGo)
    sumaGo := 0.0
    for _, nota := range notasGo {
        sumaGo += nota
        fmt.Println(nota)
    }
    promedioGo := sumaGo / float64(len(notasGo))
    fmt.Println(promedioGo)
    if promedioGo >= 3 {
        fmt.Println("Aprobado")
    } else {
        fmt.Println("Reprobado")
    }
    for indice, nota := range notasGo {
        fmt.Println(indice, nota)
    }
    fmt.Println("Fin del bloque Go")
}
